    <div id="resultDisplay">點擊下方按鈕開始抽籤</div>
    
    <button onclick="startLottery()" id="lotteryBtn">開始抽籤！</button>

    <div class="input-group">
        <input type="text" id="newFood" placeholder="輸入其他想吃的...">
        <button onclick="addFood()" style="font-size: 14px; padding: 10px;">加入清單</button>

<script>
    // 預設名單
    let foodList = [
        "學餐素食部", 
        "自己回宿舍煎素排", 
        "埔里市區素食麵", 
        "超商烤地瓜+無糖豆漿", 
        "素食滷味(不加蔥蒜)",
        "老梅打餅(素食)"
    ];

    let isRolling = false;
    let rollInterval;

    function startLottery() {
        if (foodList.length === 0) {
            alert("清單裡面沒有食物啦！請先加入選項。");
            return;
        }
        if (isRolling) return;

        isRolling = true;
        const btn = document.getElementById("lotteryBtn");
        const display = document.getElementById("resultDisplay");
        btn.innerText = "抽籤中...";
        btn.style.backgroundColor = "#ffcc00";

        // 快速輪播動畫
        let counter = 0;
        rollInterval = setInterval(() => {
            const randomIndex = Math.floor(Math.random() * foodList.length);
            display.innerText = foodList[randomIndex];
            counter++;

            // 跑大約 2 秒 (20次 * 100ms) 後停止
            if (counter > 20) {
                clearInterval(rollInterval);
                const finalIndex = Math.floor(Math.random() * foodList.length);
                display.innerText = "🎉 " + foodList[finalIndex] + " 🎉";
                btn.innerText = "再抽一次！";
                btn.style.backgroundColor = "#00d2ff";
                isRolling = false;
            }
        }, 100);
    }

    function addFood() {
        const input = document.getElementById("newFood");
        const newFood = input.value.trim();
        if (newFood !== "") {
            foodList.push(newFood);
            alert("已加入：" + newFood);
            input.value = "";
        }
    }
</script>
